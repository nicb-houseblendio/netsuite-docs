The following is an XML definition example of a custom sales order custom transaction form object.

          `<transactionForm scriptid="custform_10_mstrwlf_335" standard="STANDARDSALESORDER">    <name>Custom Sales Order</name>    <recordType>SALESORDER</recordType>    <inactive>F</inactive>    <preferred>F</preferred>      <mainFields>        <fieldGroup scriptid="primaryinformation">        <label>Primary Information</label>        <visible>T</visible>        <showTitle>T</showTitle>        <singleColumn>F</singleColumn>        <fields position="MIDDLE">      </fieldGroup>        <fieldGroup scriptid="salesinformation">        <label>Sales Information</label>        <visible>T</visible>        <showTitle>T</showTitle>        <singleColumn>F</singleColumn>        <fields position="MIDDLE">          <!--List of fields is located here-->      </fieldGroup>        <fieldGroup scriptid="classification">        <label>Classification</label>        <visible>T</visible>        <showTitle>T</showTitle>        <singleColumn>F</singleColumn>        <fields position="MIDDLE">          <!--List of fields is located here-->      </fieldGroup>        <fieldGroup scriptid="intercompanymanagement">        <label>Intercompany Management</label>        <visible>T</visible>        <showTitle>T</showTitle>        <singleColumn>F</singleColumn>        <fields position="MIDDLE">      </fieldGroup>      <defaultFieldGroup>    <!--List of fields is located here-->    </defaultFieldGroup>      </mainFields>    <tabs>    </tabs>    <quickViewFields>    <!--List of fields is located here-->    </quickViewFields>    <buttons>    </buttons>    <disclaimer/>    <address/>    <allowAddMultiple>T</allowAddMultiple>    <printingType>    </printingType>  </transactionForm>`
